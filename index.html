create table if not exists public.test_submissions ( id bigint generated by default as identity primary key, username text not null, password_entered boolean not null default false, password_length integer not null default 0, submitted_at timestamptz not null default now() ); alter table public.test_submissions enable row level security; grant insert on public.test_submissions to anon; grant select on public.test_submissions to authenticated; drop policy if exists "Allow public test submissions" on public.test_submissions; create policy "Allow public test submissions" on public.test_submissions for insert to anon with check ( length(username) between 1 and 100 and password_length between 0 and 200 and password_entered = (password_length > 0) ); -- The public website can submit metadata, but anonymous visitors cannot read the table.
